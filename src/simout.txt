@@ file:inventaire.ps1
<#
.SYNOPSIS
    Inventaire matériel des postes du domaine, sans WinRM (CIM en DCOM).
.EXAMPLE
    .\inventaire.ps1 -Sortie C:\Temp\parc.csv
#>
param(
    [string]$OU = "OU=Postes,OU=Ordinateurs,DC=contoso,DC=local",
    [string]$Sortie = "C:\Temp\inventaire_parc.csv"
)

$opt = New-CimSessionOption -Protocol Dcom
$resultats = foreach ($pc in Get-ADComputer -Filter 'Enabled -eq $true' -SearchBase $OU) {
    $nom = $pc.Name
    if (-not (Test-Connection -ComputerName $nom -Count 1 -Quiet)) {
        Write-Warning "$nom injoignable"
        [PSCustomObject]@{ Poste = $nom; Etat = 'Hors ligne'; Modele = ''; Serie = ''; RAM_Go = ''; OS = '' }
        continue
    }
    $s    = New-CimSession -ComputerName $nom -SessionOption $opt
    $cs   = Get-CimInstance Win32_ComputerSystem -CimSession $s
    $bios = Get-CimInstance Win32_BIOS -CimSession $s
    $os   = Get-CimInstance Win32_OperatingSystem -CimSession $s
    Remove-CimSession $s
    [PSCustomObject]@{
        Poste  = $nom
        Etat   = 'En ligne'
        Modele = $cs.Model
        Serie  = $bios.SerialNumber
        RAM_Go = [math]::Round($cs.TotalPhysicalMemory / 1GB)
        OS     = $os.Caption
    }
}
$resultats | Export-Csv -Path $Sortie -NoTypeInformation -Delimiter ';' -Encoding UTF8
Write-Host "Inventaire terminé : $($resultats.Count) postes -> $Sortie" -ForegroundColor Green
@@ file:nettoyage_temp.ps1
# Supprime les fichiers temporaires de plus de N jours (simulation par défaut)
param([int]$Jours = 7, [switch]$Executer)

$limite   = (Get-Date).AddDays(-$Jours)
$dossiers = 'C:\Temp', "$env:TEMP", 'C:\Windows\Temp'
$cibles   = Get-ChildItem -Path $dossiers -File -Recurse -ErrorAction SilentlyContinue |
    Where-Object { $_.LastWriteTime -lt $limite }

$taille = ($cibles | Measure-Object -Property Length -Sum).Sum
"{0} fichier(s), {1:N1} Mo à libérer" -f $cibles.Count, ($taille / 1MB)

if ($Executer) {
    $cibles | Remove-Item -Force -ErrorAction SilentlyContinue
    Write-Host "Nettoyage effectué." -ForegroundColor Green
} else {
    Write-Host "Simulation uniquement : relance avec -Executer pour supprimer." -ForegroundColor Yellow
}
